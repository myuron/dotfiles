*** eglot

    :ensure nil
    :config
    (add-to-list 'eglot-server-programs '(nix-ts-mode . ("nixd")))
    (add-to-list 'eglot-server-programs '(go-mode . ("gopls")))
    (add-to-list 'eglot-server-programs '(rust-mode . ("rust-analyzer")))
    :hook

    (go-mode . eglot-ensure)
    (rust-mode . eglot-ensure))
