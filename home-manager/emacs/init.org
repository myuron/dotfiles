*** centaur-tabs

#+begin_src emacs-lisp :tangle yes
  (use-package centaur-tabs
    :ensure nil
    :demand
    :init
    (setq centaur-tabs-style "wave")
    (setq centaur-tabs-set-bar 'under)
    :config
    (centaur-tabs-mode t)
    :bind
    ("C-<prior>" . centaur-tabs-backward)
    ("C-<next>" . centaur-tabs-forward))
#+end_src
